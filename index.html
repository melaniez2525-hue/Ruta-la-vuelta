Script de Base de Datos (SQL Editor):
create table if not exists profiles(
  id text primary key, role text not null, name text not null,
  cedula text unique not null, email text unique not null, phone text unique not null,
  address text, photo_cedula text, photo_face text, password_hash text not null,
  car text, plate text, wallet float default 0, rating float default 5,
  trips int default 0, earnings float default 0, rides int default 0,
  online boolean default false, created_at timestamptz default now()
);
create table if not exists ride_requests(
  id text primary key, passenger_id text, passenger_name text,
  passenger_rating float, passenger_phone text,
  from_name text, from_lat float, from_lng float,
  to_name text, to_lat float, to_lng float,
  km float, min int, price float, vehicle text, vehicle_icon text,
  coords jsonb, status text default 'pending',
  driver_id text, driver_name text, driver_car text, driver_plate text, driver_phone text,
  driver_lat float, driver_lng float,
  rating int, created_at timestamptz default now()
);
alter table profiles enable row level security;
alter table ride_requests enable row level security;
create policy "all_profiles" on profiles for all to anon using (true) with check (true);
create policy "all_requests" on ride_requests for all to anon using (true) with check (true);
alter publication supabase_realtime add table ride_requests;
alter publication supabase_realtime add table profiles;
